Variables CSS

Bloc lu directement dans la feuille de cette page : c'est celui qui colore tout ce qui précède. Le thème suit l'attribut data-theme posé sur <html> (piloté par Alpine et le réglage du compte), avec prefers-color-scheme en repli.

{{ css }}
/* resources/css/app.css — Tailwind v4 */
@import 'tailwindcss';
@import './voxia-tokens.css'; /* le bloc ci-contre */

@custom-variant dark (&:where([data-theme=dark], [data-theme=dark] *));

@theme inline {
  --color-fond: var(--vx-bg);
  --color-panneau: var(--vx-surface);
  --color-panneau-eleve: var(--vx-surface-2);
  --color-filet: var(--vx-line);
  --color-filet-contour: var(--vx-line-strong);
  --color-texte: var(--vx-text);
  --color-texte-secondaire: var(--vx-text-2);
  --color-texte-discret: var(--vx-text-3);
  --color-accent: var(--vx-accent);
  --color-sur-accent: var(--vx-on-accent);
  --color-accent-texte: var(--vx-accent-text);
  --color-surlignage: var(--vx-mark);
  --color-succes: var(--vx-success);
  --color-avertissement: var(--vx-warning);
  --color-erreur: var(--vx-error);
  --font-sans: var(--vx-font-text);
  --font-titre: var(--vx-font-title);
  --font-mono: var(--vx-font-data);
  --font-accent: var(--vx-font-accent);
  --radius-bouton: var(--vx-r-btn);
  --radius-champ: var(--vx-r-field);
  --radius-panneau: var(--vx-r-panel);
}

/* Les noms de classes existants (bg-fond, text-texte,
   border-filet, rounded-bouton…) restent valides :
   la migration se fait sans toucher aux vues Blade. */

@media (prefers-reduced-motion: reduce) {
  *, *::before, *::after {
    animation: none !important;
    transition: none !important;
  }
